    <script>
        // --- 1. ตั้งค่าระบบ ---
        const STATUS = "ON"; // เปลี่ยนเป็น "OFF" เพื่อแสดงหน้าปิดรับสมัคร
        
        // เชื่อมต่อ Supabase
        const supabaseUrl = 'https://dbebgpeilylrnsidform.supabase.co';
        const supabaseKey = 'sb_publishable_Y_Vppj0gv0Rh-OzmmX0gaA_Zq9Ji2SM';
        const supabase = supabase.createClient(supabaseUrl, supabaseKey);

        // จัดการหน้าจอแสดงผล
        if (STATUS === "OFF") {
            document.getElementById('formScreen').classList.add('hidden');
            document.getElementById('closedScreen').classList.remove('hidden');
        }

        // --- 2. ฟังก์ชันอัปโหลดสลิปไป Google Drive ---
        // (ต้องเอาโค้ด Apps Script ส่วนรับไฟล์ไป Deploy ใหม่ แล้วเอา URL มาใส่ตรงนี้)
        const GOOGLE_SCRIPT_URL = 'ใส่_URL_ของ_APPS_SCRIPT_ที่นี่'; 

        const toBase64 = file => new Promise((resolve, reject) => {
            const reader = new FileReader();
            reader.readAsDataURL(file);
            reader.onload = () => resolve(reader.result.split(',')[1]);
            reader.onerror = error => reject(error);
        });

        // --- 3. จัดการตอนกดปุ่ม Submit ---
        document.getElementById('orderForm').addEventListener('submit', async function(e) {
            e.preventDefault();
            
            const submitBtn = document.getElementById('submitBtn');
            submitBtn.innerText = 'กำลังประมวลผล...';
            submitBtn.disabled = true;

            try {
                // เก็บค่าจากฟอร์ม
                const formData = {
                    userType: document.getElementById('userType').value,
                    studentId: document.getElementById('studentId').value,
                    nameTitle: document.getElementById('nameTitle').value,
                    fullName: document.getElementById('fullName').value,
                    phone: document.getElementById('phone').value,
                    shirtColor: document.getElementById('shirtColor').value,
                    size: document.getElementById('shirtSize').value,
                    shippingMethod: document.getElementById('shippingMethod').value,
                };

                // โลจิกคำนวณราคาตามโค้ดเดิมของคุณ
                const priceMap = {
                    'XS':250, 'S':250, 'M':250, 'L':250, 'XL':250, '2XL':250,
                    '3XL':260, '4XL':260, '5XL':260,
                    '6XL':270, '7XL':270, '8XL':270, '9XL':270, '10XL':270
                };
                
                let pureSize = formData.size.toUpperCase().trim();
                let itemsTotal = priceMap[pureSize] || 250;
                let shippingCost = (formData.shippingMethod === "post") ? 50 : 0;
                let grandTotal = itemsTotal + shippingCost;
                let shippingLabel = (formData.shippingMethod === "post") ? "จัดส่งไปรษณีย์" : "รับที่คณะ";

                // อัปโหลดรูปลง Google Drive
                const slipFile = document.getElementById('slipFile').files[0];
                const base64Data = await toBase64(slipFile);
                
                const uploadRes = await fetch(GOOGLE_SCRIPT_URL, {
                    method: 'POST',
                    body: JSON.stringify({
                        fileName: slipFile.name,
                        mimeType: slipFile.type,
                        base64: base64Data
                    })
                });
                const uploadResult = await uploadRes.json();
                
                if(uploadResult.status !== 'success') throw new Error("อัปโหลดรูปไม่สำเร็จ");

                // บันทึกข้อมูลลง Supabase
                const { error } = await supabase.from('shirt_orders').insert([{
                    order_type: formData.userType,
                    student_id: formData.studentId || "-",
                    title: formData.nameTitle,
                    full_name: formData.fullName,
                    phone_number: formData.phone,
                    shirt_color: formData.shirtColor,
                    shirt_size: pureSize,
                    quantity: 1, // กำหนดเริ่มต้นเป็น 1
                    total_shirt_price: itemsTotal,
                    delivery_method: shippingLabel,
                    address: "-",
                    shipping_fee: shippingCost,
                    total_transfer: grandTotal,
                    slip_url: uploadResult.url
                }]);

                if (error) throw error;

                alert('บันทึกข้อมูลเรียบร้อย! ขอบคุณที่ร่วมกิจกรรมครับ');
                document.getElementById('orderForm').reset();

            } catch (error) {
                console.error(error);
                alert('เกิดข้อผิดพลาด: ' + error.message);
            } finally {
                submitBtn.innerText = 'ยืนยันการสั่งจอง';
                submitBtn.disabled = false;
            }
        });
    </script>
